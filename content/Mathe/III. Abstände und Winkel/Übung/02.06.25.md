#### a.
$$
\begin{align*}
E_1: x - y + 2z &= 7 \\
E_2: 6x + y - z &= -7
\end{align*}
$$

$$
\begin{align*}
\begin{pmatrix}1\newline -1\newline 2\end{pmatrix} &\neq r \cdot \begin{pmatrix}6\newline 1\newline -1\end{pmatrix} ~,\, r \in\mathbb R \\
\rightarrow&~\text{nicht parallel}
\end{align*}
$$

$$
\begin{align*}
\text{I.}&& x - y + 2z &= 7 \\
\text{II.}&& 6x + y - z &= -7 \\
\text{I.} + \text{II.}&& 7x + z &= 0 \\
\rightarrow&& z &= -7x \\
\rightarrow&& x - y - 14x &= 7 \\
\rightarrow&& y &= -7-13x \\
\Rightarrow&& L &= \left\{ t\in\mathbb R; x=t; y = -7-13x; z=-7x \right\} \\
\Rightarrow && g: \overrightarrow{ x } &= \begin{pmatrix}0\newline -7\newline 0\end{pmatrix} + t \begin{pmatrix}1\newline -13\newline -7\end{pmatrix}
\end{align*}
$$
