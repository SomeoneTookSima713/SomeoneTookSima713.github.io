#### b)
$$
\begin{align*}
f(x) &= \frac12e^x+2x \\
f'(x) &= \frac12e^x+2 \\
x_0 &:= 0 \\
t: y &= mx + n \\
m &= f'(x_0) \\
&= f'(0) \\
&= 2,5 \\
y &= f(x_0) \\
&= f(0) \\
&= 0,5 \\
y &= mx+n \\
n &= y-mx \\
&= 0,5-2,5\cdot0 \\
&= 0,5 \\
\Rightarrow t: y &= 2,5x+0,5
\end{align*}
$$

#### e)
$$
\begin{align*}
f(x) &= 3\ln x \\
f'(x) &= \frac3x \\
x_0 &:= e \\
t: y &= mx + n \\
m &= f'(x_0) \\
&= f'(e) \\
&= \frac3e \\
y &= f(x_0) \\
&= f(e) \\
&= 3 \\
y &= mx+n \\
n &= y-mx \\
&= 3-\frac3e\cdot e \\
&= 0 \\
\Rightarrow t: y &= \frac3ex
\end{align*}
$$

#### f)
$$
\begin{align*}
f(x) &= 2\ln x + 4x \\
f'(x) &= \frac2x + 4 \\
x_0 &:= 1 \\
t: y &= mx + n \\
m &= f'(x_0) \\
&= f'(1) \\
&= 6 \\
y &= f(x_0) \\
&= f(1) \\
&= 4 \\
y &= mx+n \\
n &= y-mx \\
&= 4-6\cdot1 \\
&= -2 \\
\Rightarrow t: y &= 6x-2
\end{align*}
$$
