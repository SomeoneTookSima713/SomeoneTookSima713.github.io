#### a)
$\displaystyle f(x) = e^x $
$\displaystyle f'(x) = e^x = f(x) $

$\displaystyle A(1|e) $
$\displaystyle B(-1|e^{-1}) $

$\displaystyle t: y=mx+n $

$$
\begin{align*}
t_A: y &= m_Ax+n_A \\
m_A &= f'(x_A) \\
&= e \\
y_A &= m_Ax_A + n_A \\
n_A &= m_Ax_A-y_A \\
&= e\cdot1 - e \\
&= 0 \\
\Rightarrow t_A: y &= ex
\\~\\
t_B: y &= m_Bx+n_B \\
m_B &= f'(x_B) \\
&= \frac1e \\
y_B &= m_Bx_B + n_B \\
n_B &= m_Bx_B-y_B \\
&= \frac1e \cdot -1 - \frac1e \\
&= \frac2e \\
\Rightarrow t_B: y &= \frac {x+2}e
\end{align*}
$$
