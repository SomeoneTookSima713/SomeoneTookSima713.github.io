#### a. $\displaystyle f_t(x) = xe^{-tx} $

$$
\begin{align*}
f_{t1}(x) &= f_{t2}(x) &t_1\neq t_2 \\
xe^{-ax} &= xe^{-bx} &|~~ x &= 0\\
0\cdot e^0 &= 0\cdot e^0 &\text{w.A.}\\
\rightarrow S(0|0) \\
xe^{-ax} &= xe^{-bx} &x\neq 0 &~\land~ t_1\neq t_2 \\
-ax &= -bx &\text{f.A.}\\
\end{align*}
$$

$\displaystyle S(0|0) $

#### b. $\displaystyle f_t(x) = 2(x+5)e^{tx} $

$$
\begin{align*}
f_{t1}(x) &= f_{t2}(x) &t_1\neq t_2 \\
2(x+5)e^{t_1x} &= 2(x+5)e^{t_2x} &|~~x = -5 \\
0e^0 &= 0e^0 &\text{w.A.} \\
\rightarrow S&(0|0) \\
2(x+5)e^{t_1x} &= 2(x+5)e^{t_2x} &|~~x = -5 &~\land~ t_1\neq t_2 \\
t_1x &= t_2x &\text{f.A.}
\end{align*}
$$

$\displaystyle S(0|0) $

#### c. $\displaystyle f_t(x) = e^{t(x-2)}x+3 $

$$
\begin{align*}
f_{t1}(x) &= f_{t2}(x) &t_1\neq t_2 \\
e^{t_1(x-2)}x+3 &= e^{t_2(x-2)}x+3 &|~~ x = 0 \\
3 &= 3 &\text{w.A.} \\
\rightarrow S_1&(0|3) \\
e^{t_1(x-2)}x+3 &= e^{t_2(x-2)}x+3 &|~~ x \neq 0 &~\land~ t_1\neq t_2 \\
t_1(x-2) &= t_2(x-2) \\
\rightarrow x &= 2 \\
\rightarrow S_2&(2|5)
\end{align*}
$$

$\displaystyle S_1(0|3); S_2(2|5) $
