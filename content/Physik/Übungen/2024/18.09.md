## LB. S. 92
### Nr. 11.

$$
\begin{align*}
m &:= 1100 \text{ kg} \\
v &:= 65 \frac{\text{km}}{\text h} = 18,0\overline5 \frac{\text m}{\text s} \\
s &:= 55 \text{ m}
\end{align*}
$$

$$
\begin{align*}
s &= \frac a2t^2 \\
v &= at \\
\rightarrow s &= \frac a2 \left( \frac va \right)^2 \\
s &= \frac{v^2}{2a} \\
55 \text{ m} &= \frac{2.640.625 \text{ m}^2}{180 \text{ s}^2 \cdot a} \\
a &= \frac{2.640.625 \text{ m}^2}{9900 \text{ s}^2\text m} \\
&\approx 266,730 \frac{\text m}{\text s^2}
\end{align*}
$$

### Nr. 15.

$$
\begin{align*}
\overrightarrow F &:= 270 \text{ kN} \\
m &:= 500 \text{ t} = 500.000 \text{ kg}
\end{align*}
$$

#### a)

$$
\begin{align*}
\overrightarrow F &= m \cdot a \\
a &= \frac{\overrightarrow F}{m} \\
&= \frac{270.000 \text{ N}}{500.000 \text{ kg}} \\
&= 0,54 \frac{\text m}{\text s^2}
\end{align*}
$$

#### b)

$$
\begin{align*}
v &:= 100 \frac{\text{km}}{\text h} &= \frac{250}{9} \frac{\text m}{\text s}
\end{align*}
$$

$$
\begin{align*}
v &= a \cdot t \\
t &= \frac va \\
&= \frac{\frac{250}{90} \frac{\text m}{\text s}}{0,54 \frac{\text m}{\text s^2}} \\
&\approx 51,4 \text{ s}
\end{align*}
$$

#### c)

$$
\begin{align*}
a &:= 0,60 \frac{\text m}{\text s^2} \\
v &:= 200 \frac{\text{km}}{\text h} = \frac{500}{9} \frac{\text m}{\text s}
\end{align*}
$$

$$
\begin{align*}
v &= a \cdot t \\
t &= \frac va \\
&= \frac{\frac{500}{9} \frac{\text m}{\text s}}{0,60 \frac{\text m}{\text s^2}} \\
&= 92,593 \text{ s} \\
s &= \frac{v^2}{2a} \\
&\approx 2572 \text{ m} \\
\overrightarrow F &= -a \cdot m \\
&= -0,60 \frac{\text m}{\text s^2} \cdot 500.000 \text{ kg} \\
&= -300.000 \text{ N}
\end{align*}
$$
