$$
\begin{align*}
U_0 &\colonequals 4,5 \text{ V} \\
R &\colonequals 500 \text{ k}\Omega \\
C &\colonequals 5,0 \text{ mF} \\
U_\text{min} &\colonequals 3,3 \text{ V}
\end{align*}
$$

$$
\begin{align*}
U(t) &= U_0 \cdot e ^ {\displaystyle -\frac t{RC}} \\
&\geq U_\text{min} \\
\rightarrow U_0 e^{\displaystyle -\frac t{RC}} &\geq U_\text{min} \\
\rightarrow -\frac{t}{RC} &\geq \ln\left(\frac{U_\text{min}}{U_0}\right) \\
\rightarrow t &\leq -RC\ln\left( \frac{U_\text{min}}{U_0} \right) \\
&\leq RC\ln\left( \frac{U_0}{U_\text{min}} \right) \\
\Rightarrow t_\text{max} &\approx 775,39 \text{ s} 
\end{align*}
$$
