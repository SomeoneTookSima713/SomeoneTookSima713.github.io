$\displaystyle
h := 3\text{ m} \\
m := 2\text{ kg} \\
v := 4\frac{\text m}{\text s}
$

$$
\begin{align*}
E_\text{pot} &= mgh \\
&= 2\text{ kg} \cdot 9,81\frac{\text m}{\text s^2} \cdot 3\text{ m} \\
&= 58,86\text{ J}
\\~\\
E_\text{kin} &= \frac12mv^2 \\
&= \frac12 \cdot 2\text{ kg} \cdot \left(4\frac{\text m}{\text s}\right)^2 \\
&= 16\text{ J}
\\~\\
E_\text{pot} &\neq E_\text{kin} \\
\Rightarrow ~~&\text{Der Energieerhaltungssatz}\\&\text{der Mechanik gilt nicht}
\end{align*}
$$

$$
\begin{align*}
W_\text{Reibung} &= E_\text{pot} - E_\text{kin} \\
&= (58,86 - 16)\text{ J} \\
&= 42,86\text{ J}
\end{align*}
