### a)

$
\begin{array}{lll}
E_\text{pot} &-& \text{potentielle Energie im ausgelenkten Zustand} \\
t &-& \text{benötigte Zeit, um vom ausgelenkten Zustand die Ruhelage zu erreichen}
\end{array}
$

$
l := 1,5\text{ m} \\
\alpha := 20\degree \\
m = 200\text{ g}
$

$$
\begin{align*}
h &= l\cdot \cos \alpha\\
&= 1,5\text{ m} \cdot \cos(20\degree) \\
&\approx 1,410\text{ m}\\
E_\text{pot} &= m \cdot g \cdot h \\
&\approx 0,2\text{ kg} \cdot 9,81\frac{\text m}{\text s^2} \cdot 1,410\text{ m} \\
&\approx \boxed{2,766\text{ J}}
\\~\\
T &= 2\pi\sqrt{\frac lg} \\
&= 2\pi\sqrt{\frac{1,5\text{ m}}{9,81\frac{\text m}{\text s^2}}} \\
&\approx 2,457 \text{ s}
\\~\\
t &= \frac T4 \\
&\approx \boxed{0,614\text{ s}}
\end{align*}
$$

### b)

$
\begin{array}{lll}
E_{\text{pot}0} &-& \text{Potentielle Energie zu Beginn des Falls} \\
E_{\text{kin}1} &-& \text{Kinetische Energie beim Aufprall}
\end{array}
$

$
m := 70 \text{ kg} \\
h := 15\text{ m} \\
$

$$
\begin{align*}
E_{\text{pot}0} &= m \cdot g \cdot h \\
&= 70\text{ kg} \cdot 9,81\frac{\text m}{\text s^2} \cdot 15\text{ m} \\
&= 10.300,5\text{ J}
\\~\\
E_{\text{kin}1} &= E_{\text{pot}0}\\
&= \frac12mv^2 \\
\rightarrow \frac12mv^2 &= E_{\text{pot}0}
\\~\\
\Rightarrow v &= \sqrt{2E_{\text{pot}0}\over m} \\
&= \sqrt{2\cdot 10.300,5\text{ J} \over{70 \text{ kg}}} \\
&\approx \boxed{17,155 \frac{\text{m}}{\text{s}}}
\end{align*}
$$

### c)

$
m := 70\text{ g} = 0,07\text{ kg} \\
v := 30\frac{\text m}{\text s} \\
s := 30\text{ cm} = 0,3\text{ m}
$

$$
\begin{align*}
E_{\text{kin}} &= \frac12mv^2 \\
&= \frac12\cdot 0,07\text{ kg} \cdot \left(30\frac{\text m}{\text s}\right)^2 \\
&= 31,5\text{ J}
\\~\\
E_{\text{pot}} &= E_{\text{kin}} \\
&= \frac12Ds^2 \\
\rightarrow \frac12Ds^2 &= E_{\text{kin}}
\\~\\
\Rightarrow D &= {2E_{\text{kin}} \over {s^2}} \\
&= {2\cdot 31,5\text{ J} \over {\left(0,3\text{ m}\right)^2}} \\
&= \boxed{700\frac{\text N}{\text m}}
\end{align*}
$$
