### Experiment 2

@import "../Bilder/Experiment2.jpeg"

$$
\begin{align*}
m_1 &:= 48 \text{ g}\\
m_2 &:= 20 \text{ g}\\
s &:= 0,5 \text{ m}\\
s_f &:= 0,2 \text{ m} \\
h &:= 0,145 \text{ m}\\
\alpha &= \arcsin\frac{h}{s} \\
&\approx 16,9\degree
\end{align*}
$$

$$
\begin{align*}
F_G - F_R - F_\text{Hang} &= (m_1+m_2)a \\
m_2g - F_R - m_1g\sin\alpha &= \frac{ 2(m_1+m_2)s_f }{ t^2 } \\
% \rightarrow F_R &= -\frac{ 2(m_1+m_2)s_f }{ t^2 } + (m_2 - m_1\sin\alpha)g
\rightarrow F_R &= \frac{-g \; m_1 \; t^{2} \; \operatorname{sin} \left( \alpha \right) + g \; m_2 \; t^{2} - 2 \; m_1 \; s_f - 2 \; m_2 \; s_f}{t^{2}}
\end{align*}
$$

$$
\begin{align*}
t &:= 0,8 \text{ s} \\
F_R &\approx 0,29 \text{ N}
\end{align*}
$$

## $~$

#### Zusatz: Endgeschwindigkeit
###### Annahme: Die Masse $m_2$ liegt zu Versuchsbeginn exakt $20 \text{ cm}$ über dem Boden
$$
\begin{align*}
F_R &= \frac{-g \; m_1 \; t^{2} \; \operatorname{sin} \left( \alpha \right) + g \; m_2 \; t^{2} - 2 \; m_1 \; s_f - 2 \; m_2 \; s_f}{t^{2}} \\
\rightarrow \frac{-F_R \; t - g \; m_1 \; t \; \operatorname{sin} \left( \alpha \right) + g \; m_2 \; t}{m_1 + m_2} &= \frac{ 2s_f }{ t } \\
&= v \\
&\approx 2,91 \frac{\text m}{\text s}
\end{align*}
$$
