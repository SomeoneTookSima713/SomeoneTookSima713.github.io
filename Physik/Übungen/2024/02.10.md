$$
\begin{align*}
m &:= 120 \text{ g} = 0,12 \text{ kg} \\
\alpha &:= 10 \degree \\
s &:= 40 \text{ cm} = 0,4 \text{ m} \\
v_1 &:= 0,8 \frac{\text m}{\text s} \\
h &:= 6,95 \text{ cm} = 0,0695\text{ m}
\end{align*}
$$

$$
\begin{align*}
E_\text{pot} &= mgh \\
E_\text{kin} &= \frac12mv^2 \\
W_R &= E_\text{pot} - E_\text{kin} \\
&= mgh - \frac12mv^2 \\
F_R &= \frac{W_R}{s} \\
&= \frac{mgh - \frac12mv^2}{s} \\
F_N &= -F_G \cdot \cos\alpha \\
F_R &= \mu \cdot F_N \\
\rightarrow \mu &= \frac{F_R}{F_N} \\
&= \frac{mgh - \frac12mv^2}{sF_G\cos\alpha} \\
&= \frac{h}{s\cos\alpha}-\frac{v^2}{2gs\cos\alpha} \\
&= \frac{0,0695 \text{ m}}{0,4\text{ m} \cdot \cos(10\degree)} - \frac{\left(0,8 \frac{\text m}{\text s}\right)^2}{2\cdot9,81 \frac{\text m}{\text s^2} \cdot 0,4 \text{ m} \cdot \cos(10°)} \\
&\approx 0,094
\end{align*}
$$
