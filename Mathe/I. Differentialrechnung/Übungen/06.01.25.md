## 1.

$$
\begin{align*}
f(x) &= x^4 - kx^2 &k\in\mathbb R^+
\end{align*}
$$

### a.
$$
\begin{align*}
f'(x) &= \lim_{h\to0} \frac{f(x+h)-f(x)}{h} \\
&= \lim_{h\to0} \frac{(x+h)^4 - k(x+h)^2 - x^4 + kx^2}{h} \\
&= \lim_{h\to0} \frac{(x^2+2xh+h^2)^2 - kx^2+2kxh+kh^2 - x^4 + kx^2}{h} \\
&= \lim_{h\to0} \frac{x^4+4x^3h+6x^2h^2 + 4xh^3 + h^4 - kx^2-2kxh-kh^2 - x^4 + kx^2}{h} \\
&= \lim_{h\to0} \frac{4x^3h+6x^2h^2 + 4xh^3 + h^4 - 2kxh - kh^2}{h} \\
&= \lim_{h\to0} 4x^3+6x^2h + 4xh^2 + h^3 - 2kx - kh \\
&= 4x^3 - 2kx \\
&= 2x\cdot(2x^2-k)
\end{align*}
$$
